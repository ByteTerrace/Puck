# The feedback pipeline compiles and installs; paused and reset, submission 2 holds 2/16.
pipeline.wait feedback compiled 40

pipeline.time feedback pause

pipeline.step feedback
pipeline.wait feedback submitted 2 10
# A 256 KiB cap. The installed graph holds 52224 bytes: three frame slots of a 32x32 half-float history (24576), the
# 32x32 RGBA8 gray (12288), the copy's 32x32 RGBA8 targets (12288), and four 256-byte constant buffers, the frame
# group's and each pass's (3072).
pipeline.budget feedback 262144
pipeline.capture feedback {run}/before.png
pipeline.wait feedback captured 10
# The capture gave the readback a 32x32 RGBA8 staging buffer (4096), so the instance owns 56320. The wide edit grows
# the history to 256x256: 1600512 bytes steady-state, which beside those 56320 peak at 1656832, over the cap, so the
# candidate is refused before it allocates anything. A reload is not a step, and the
# paused instance still presents submission 2.
pipeline.load feedback wide.graph.json
pipeline.status
pipeline.wait feedback installed 40
pipeline.capture feedback {run}/after-refusal.png
pipeline.wait feedback captured 10
# The installed graph keeps running: submission 3 holds 3/16 over its retained history.
pipeline.step feedback
pipeline.wait feedback submitted 3 10
pipeline.capture feedback {run}/refused-step.png
pipeline.wait feedback captured 10
# An in-budget edit, one minus the history, carries the retained history instead of allocating it, so it peaks at
# 56320 + 52224 - 24576 = 83968 and installs while paused; the next step renders submission 4 through it over the
# retained history as 1 - 4/16.
pipeline.load feedback ../pipeline-edit/corrected.graph.json
pipeline.status
pipeline.wait feedback installed 40
pipeline.step feedback
pipeline.wait feedback submitted 4 10
pipeline.capture feedback {run}/in-budget.png
