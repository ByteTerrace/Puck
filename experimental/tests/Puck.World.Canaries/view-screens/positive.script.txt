# Every frame renders every pass, and every produced frame refreshes the diegetic views.
# The engine ready first: its pipeline set installed and its first frame produced, however long a cold driver
# cache takes.
world.wait ready 45
world.cadence off
world.view-refresh 1
world.wait 30
# Both screens show camera 'eye': one camera view is registered.
world.view-refresh
world.counters gpu
world.wait 30
# Thirty ticks later the eye view has rendered once for every world frame, not once per screen.
world.counters gpu
world.screenshot {run}/frame.png
world.wait 10
# The room's view moves into a centred three-quarter slot, so it is a pane a display point walks into. A lone view over
# the whole display is the display itself at any render scale: the root stands for it and publishes it as no pane.

world.wait 10

world.screens
