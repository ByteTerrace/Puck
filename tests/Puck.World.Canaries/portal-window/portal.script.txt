# The door's glass face shows a window session onto its destination.
world.screens

# The room's view moves into a centred three-quarter slot, so it is a pane a display point walks into. A lone view over
# the whole display is the display itself at any render scale: the root stands for it and publishes it as no pane.
world.row.set views.layouts {"name":"doorway","seatCount":0,"slots":[{"height":0.75,"width":0.75,"x":0.125,"y":0.125}],"transitionRenderScale":1,"transitionSeconds":0}
