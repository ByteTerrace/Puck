# The boot row names the canary's own tint source; it compiles in the background.
pipeline.wait tint compiled 40
pipeline.wait tint installed 10
# A source row: preview gain 0.25, commit it into the row, and capture the one initialization frame after a reset.
pipeline.set tint tint {"gain":0.25}
pipeline.commit tint
pipeline.overrides tint
pipeline.reset tint
pipeline.wait tint submitted 1 10
pipeline.capture tint {run}/source.png
pipeline.wait tint captured 10
# The same instance names the relocated package, whose source copy the runner deleted, and carries the committed
# override, which the server binds against the package's config schema.
world.row.set views.graphs {"name":"tint","source":"{run}/tint","overrides":{"tint":{"gain":0.25}}}
pipeline.reload tint
pipeline.wait tint compiled 40
pipeline.wait tint installed 10
pipeline.overrides tint
pipeline.reset tint
pipeline.wait tint submitted 1 10
pipeline.capture tint {run}/package.png
pipeline.wait tint captured 10
world.save {run}/package.world.json
# The same package at the high tier: the host loads the package's high variant, compiled with PUCK_QUALITY_TIER
# defined, so the blue reads 0.75 where the variant no tier names reads 0.5.
world.row.set views.graphs {"name":"tint","source":"{run}/tint","overrides":{"tint":{"gain":0.25}},"tier":"high"}
pipeline.reload tint
pipeline.wait tint compiled 40
pipeline.wait tint installed 10
pipeline.status
pipeline.reset tint
pipeline.wait tint submitted 1 10
pipeline.capture tint {run}/tiered.png
pipeline.wait tint captured 10
# The low tier, which the graph does not declare, falls back to the default variant, and pipeline.status says so.
world.row.set views.graphs {"name":"tint","source":"{run}/tint","overrides":{"tint":{"gain":0.25}},"tier":"low"}
pipeline.reload tint
pipeline.wait tint compiled 40
pipeline.wait tint installed 10
pipeline.status
pipeline.reset tint
pipeline.wait tint submitted 1 10
pipeline.capture tint {run}/fallback.png
pipeline.wait tint captured 10
